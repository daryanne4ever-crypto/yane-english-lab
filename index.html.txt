    <script>
        // JAVASCRIPT UNIFICADO
        function login() {
            const user = document.getElementById('username').value;
            if (user.trim() === "") {
                alert("Please, tell us your name first!");
                return;
            }

            document.getElementById('login-section').style.display = 'none';
            document.getElementById('activity-section').style.display = 'block';
            document.getElementById('user-greeting').innerText = `Hello, ${user}! Ready to practice?`;
        }

        // Web Speech API
        const startBtn = document.getElementById('start-btn');
        const btnText = document.getElementById('btn-text');
        const micIcon = document.getElementById('mic-icon');
        const output = document.getElementById('speech-output');

        const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;

        if (SpeechRecognition) {
            const recognition = new SpeechRecognition();
            recognition.lang = 'en-US'; 
            recognition.interimResults = false;

            startBtn.addEventListener('click', () => {
                try {
                    recognition.start();
                } catch (e) {
                    console.log("Recognition already started");
                }
            });

            recognition.onstart = () => {
                btnText.innerText = "Listening...";
                startBtn.style.background = "#22c55e"; // Green
                micIcon.innerText = "⭕"; // Indicador de gravando
            };

            recognition.onresult = (event) => {
                const transcript = event.results[0][0].transcript;
                output.innerText = `"${transcript}"`;
                output.style.color = "#1e293b";
                output.style.fontStyle = "normal";
            };

            recognition.onspeechend = () => {
                recognition.stop();
                resetButton();
            };

            recognition.onerror = (err) => {
                resetButton();
                if(err.error !== 'no-speech') {
                    output.innerText = "Error: " + err.error;
                }
            };

            function resetButton() {
                btnText.innerText = "Start Speaking";
                startBtn.style.background = "#ef4444"; // Red
                micIcon.innerText = "🎤";
            }

        } else {
            output.innerText = "Sorry, your browser does not support voice recognition. Try Chrome or Edge.";
        }
    </script>
